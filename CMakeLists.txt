cmake_minimum_required(VERSION 3.20)

# Pull in ARM toolchain and MCU flags before project() call
set(CMAKE_TOOLCHAIN_FILE ${CMAKE_SOURCE_DIR}/cmake/arm-none-eabi.cmake)
include(cmake/stm32g4.cmake)

# FreeRTOS kernel source files
set(FREERTOS_SOURCES
    FreeRTOS/Source/tasks.c
    FreeRTOS/Source/queue.c
    FreeRTOS/Source/list.c
    FreeRTOS/Source/timers.c
    FreeRTOS/Source/event_groups.c
    FreeRTOS/Source/portable/GCC/ARM_CM4F/port.c
    FreeRTOS/Source/portable/MemMang/heap_4.c
)

# FreeRTOS header locations
include_directories(
    FreeRTOS/Source/include
    FreeRTOS/Source/portable/GCC/ARM_CM4F
)




project(bldc-motor-controller C ASM)
set(CMAKE_C_STANDARD 11)

# Linker script generated by CubeMX
set(LINKER_SCRIPT ${CMAKE_SOURCE_DIR}/Core/Startup/STM32G431CBUX_FLASH.ld)

# All HAL and CMSIS source files
file(GLOB_RECURSE HAL_SOURCES
    Drivers/STM32G4xx_HAL_Driver/Src/*.c
)

# CubeMX generated core files
set(CORE_SOURCES
    Core/Src/main.c
    Core/Src/stm32g4xx_it.c
    Core/Src/stm32g4xx_hal_msp.c
    Core/Src/system_stm32g4xx.c
    Core/Startup/startup_stm32g431cbux.s
)

# Your application code
file(GLOB APP_SOURCES App/Src/*.c)

# Header locations
include_directories(
    Core/Inc
    Drivers/STM32G4xx_HAL_Driver/Inc
    Drivers/STM32G4xx_HAL_Driver/Inc/Legacy
    Drivers/CMSIS/Device/ST/STM32G4xx/Include
    Drivers/CMSIS/Include
    App/Inc
)

# Build the firmware binary
add_executable(${PROJECT_NAME}.elf
    ${CORE_SOURCES}
    ${HAL_SOURCES}
    ${APP_SOURCES}
    ${FREERTOS_SOURCES}
)

# Apply linker script
target_link_options(${PROJECT_NAME}.elf PRIVATE
    -T${LINKER_SCRIPT}
)

# Generate .bin file from .elf after build
add_custom_command(TARGET ${PROJECT_NAME}.elf POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O binary
    ${PROJECT_NAME}.elf ${PROJECT_NAME}.bin
    COMMENT "Generating .bin file"
)
